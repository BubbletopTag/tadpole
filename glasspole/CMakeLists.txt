# Glasspole.
#
# CMake rather than the Makefile the rest of Tadpole uses, for two reasons that
# both come from outside: dynarmic is a CMake project, and the Windows build
# will be a MinGW cross-compile driven by a toolchain file. Fighting either of
# those with hand-written rules is not a good use of anyone's afternoon.
cmake_minimum_required(VERSION 3.20)
project(glasspole C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# ---- dynarmic -------------------------------------------------------------
set(DYNARMIC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/deps/dynarmic")
if(NOT EXISTS "${DYNARMIC_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "dynarmic is missing — run ./fetch-deps.sh first")
endif()

# A32 only: the guest is 32-bit ARM and the A64 frontend is a third of the
# build for code we will never execute.
set(DYNARMIC_FRONTENDS "A32" CACHE STRING "" FORCE)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
# dynarmic's bundled externals still declare cmake_minimum_required(VERSION 3.x)
# below what CMake 4 will accept without this.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
add_subdirectory("${DYNARMIC_DIR}" EXCLUDE_FROM_ALL)

# ---- the host backend -----------------------------------------------------
# Exactly one of these is compiled. That is the whole design: if a second file
# ever needs an #ifdef to choose between them, the interface in host.h is wrong.
if(WIN32)
    set(GP_HOST src/host_win32.c)
else()
    set(GP_HOST src/host_posix.c)
endif()

# The icon, on Windows only. windres does not exist in a Linux build and a .rc
# means nothing to ELF, so the resource joins the source list rather than being
# compiled unconditionally. CMAKE_RC_COMPILER comes from the cross toolchain
# file tools/build-windows.sh writes.
set(GP_RC)
if(WIN32)
    enable_language(RC)
    set(GP_RC src/glasspole.rc)
endif()

add_executable(glasspole src/cpu.cpp src/syscall.cpp src/signal.cpp src/cp15.cpp
                         src/kuser.cpp src/elf.c src/path.cpp ${GP_HOST} ${GP_RC})
target_include_directories(glasspole PRIVATE src)
target_link_libraries(glasspole PRIVATE dynarmic)

# WHICH BUILD THIS IS. glasspole.exe carried no version at all — it is one of
# the three executables the Windows installer ships, and the only one a user is
# at all likely to run by hand, so "what version is this?" had no answer for it
# even after tadpole-view.exe learned to report one. Nothing could verify it
# either: the release script checks its assets by looking for the version
# string inside them, and a binary that never contained one could not be
# checked, only assumed.
#
# A GENERATED FILE RATHER THAN A -D, for two reasons. A define is only baked
# into a binary if some translation unit actually uses it, and adding a use to
# a hand-written source would put the version in a file that has nothing to do
# with versioning. And file(GENERATE) rewrites only when the content differs,
# so an unchanged version does not force a rebuild while a changed one does —
# the same property tadpole/Makefile needs .tadpole-version for, obtained here
# for free.
set(TADPOLE_VERSION "dev" CACHE STRING
    "Release version baked into glasspole, as DDMMYYYY-NNNN")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/gp_version.c"
     CONTENT "/* Generated by CMake. The release scripts read this string back\n   out of the finished binary with `strings`. */\nconst char gp_build_version[] = \"${TADPOLE_VERSION}\";\n")
target_sources(glasspole PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/gp_version.c")
if(WIN32)
    # bcrypt: gp_random. winmm: timeBeginPeriod, so nanosleep is ~1 ms rather
    # than ~15.6. Both are system DLLs on every Windows since 7.
    target_link_libraries(glasspole PRIVATE bcrypt winmm)
    # Static runtime: one file someone can copy, with no MinGW DLLs beside it.
    target_link_options(glasspole PRIVATE -static)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(glasspole PRIVATE Threads::Threads)
    # THE C++ RUNTIME COMES WITH IT, for the same reason -static does on
    # Windows: this binary now ships inside the AppImage as the default engine,
    # and it is built on whatever distribution the release was rolled on.
    #
    # The alternative was copying the build machine's libstdc++.so.6 into the
    # image's usr/lib, and that is the AppImage mistake everyone makes once: the
    # bundled library goes in front of the host's for EVERY process in the
    # image, including the GL driver the viewer loads, and a mesa built against
    # a newer libstdc++ than the one we shipped fails to load with a GLIBCXX_3.4
    # version error that names neither Tadpole nor the library it came from.
    # Statically linking the runtime into the one binary that needs it leaves
    # the host's own libraries alone.
    target_link_options(glasspole PRIVATE -static-libstdc++ -static-libgcc)
endif()

# ---- the host-side tests ---------------------------------------------------
# One executable, no framework, run by ctest or by hand. It links path.cpp
# against whichever host backend this build uses and needs neither dynarmic nor
# a guest, which is the point: the path rules are the one part of the emulator
# whose worst bug (a drive letter on Windows) cannot be caught by diffing
# against qemu-arm, because qemu-arm does not run there.
enable_testing()
add_executable(gp-path-test tests/path_test.cpp src/path.cpp ${GP_HOST})
target_include_directories(gp-path-test PRIVATE src)
if(WIN32)
    target_link_libraries(gp-path-test PRIVATE bcrypt winmm)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(gp-path-test PRIVATE Threads::Threads)
endif()
add_test(NAME path COMMAND gp-path-test)

# ---- the guest-side test programs -----------------------------------------
# Built with clang's ARM target, matching the flags tadpole/Makefile already
# uses for the shim: ARMv7-A, VFP, softfp — i.e. the real console.
find_program(GP_CLANG clang)
find_program(GP_OBJCOPY NAMES llvm-objcopy objcopy)

if(GP_CLANG AND GP_OBJCOPY)
    # -fuse-ld=lld because the host's GNU ld is built for x86 only and answers
    # an ARM link with "unrecognised emulation mode: armelf_linux_eabi". lld is
    # multi-target by construction, so this needs no cross-binutils install.
    set(GP_ARMFLAGS --target=armv7-unknown-linux-gnueabi
                    -march=armv7-a -mfpu=vfp -mfloat-abi=softfp -fuse-ld=lld)
    foreach(gp_test hello thread threads-many abort segv)
    add_custom_command(
        OUTPUT  ${CMAKE_CURRENT_BINARY_DIR}/${gp_test}.bin
        COMMAND ${GP_CLANG} ${GP_ARMFLAGS} -nostdlib -static
                -Wl,-Ttext=0x00010000 -Wl,-e,_start
                -o ${CMAKE_CURRENT_BINARY_DIR}/${gp_test}.elf
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/${gp_test}.S
        # -j .text, and it matters. Without it objcopy also carries
        # .ARM.attributes, which sits at VMA 0, so the flat binary comes out
        # with 64 KB of zeros ahead of the code — and those zeros decode as
        # `andeq r0, r0, r0`. The guest then runs a 16,384-instruction NOP sled
        # and falls into _start by luck. It works, which is the problem.
        COMMAND ${GP_OBJCOPY} -O binary -j .text
                ${CMAKE_CURRENT_BINARY_DIR}/${gp_test}.elf
                ${CMAKE_CURRENT_BINARY_DIR}/${gp_test}.bin
        DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/${gp_test}.S
        COMMENT "Assembling the ARM guest test program ${gp_test}")
    list(APPEND gp_test_bins ${CMAKE_CURRENT_BINARY_DIR}/${gp_test}.bin)
    endforeach()
    add_custom_target(gp-tests ALL DEPENDS ${gp_test_bins})
else()
    message(WARNING "clang or objcopy missing — guest test programs will not build")
endif()
